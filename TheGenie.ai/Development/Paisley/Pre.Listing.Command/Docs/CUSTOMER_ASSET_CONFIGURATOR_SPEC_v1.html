CTA System Architecture

The CTA Strategy

Separate CTA design & selection (our scope) from CTA execution & tracking (PLS/Genie Application team). We focus on building amazing CTAs and configuring which ones appear where. They handle engagement center integration.

📤 Two Deployment Paths

⚡ Automated (FarmCast)

System-triggered marketing campaigns

  • Listing Command - New listing triggers kit
  • Neighborhood Command - Market updates
  • Comp Command - Competition analysis
  • CTAs auto-configured from Customer Config
  • Engagement Center tracks all interactions
SMS Link → Landing Page → CTA → Lead Capture → Engagement Center

📋 Non-Automated (Static)

Human-distributed marketing materials

  • Door-to-door flyers with QR codes
  • Open House handouts
  • Print brochures with landing page URLs
  • Same CTA config applies
  • Entry point differs (QR vs SMS link)
QR Code → Landing Page → CTA → Lead Capture → Engagement Center

🗄️ Proposed Database Schema

-- User-level service configuration (maps service → collection → landing page)
CREATE TABLE FarmGenie.dbo.UserServiceCollection (
    UserServiceCollectionId INT IDENTITY(1,1) PRIMARY KEY,
    AspNetUserId NVARCHAR(128) NOT NULL,
    PropertyCastTypeId INT NOT NULL,        -- 1=LC, 2=NC, 3=CC, 4=PLS
    CollectionName NVARCHAR(100) NOT NULL,    -- "dainelle-listing-kit"
    LandingPageAsset NVARCHAR(100),           -- "landing-pages/lc-hollywood"
    HubAssetSettingId INT,                    -- Links to existing system
    IsActive BIT DEFAULT 1,
    CreateDate DATETIME DEFAULT GETDATE(),
    LastModified DATETIME DEFAULT GETDATE(),
    ModifiedBy NVARCHAR(128),                 -- Admin who made the change
    
    CONSTRAINT FK_UserServiceCollection_User 
        FOREIGN KEY (AspNetUserId) REFERENCES AspNetUsers(Id),
    CONSTRAINT UQ_UserServiceCollection 
        UNIQUE (AspNetUserId, PropertyCastTypeId)
);

-- Index for fast lookups by user
CREATE INDEX IX_UserServiceCollection_UserId 
    ON UserServiceCollection(AspNetUserId);

-- CTA Configuration per landing page
CREATE TABLE FarmGenie.dbo.LandingPageCtaConfig (
    LandingPageCtaConfigId INT IDENTITY(1,1) PRIMARY KEY,
    AspNetUserId NVARCHAR(128),             -- NULL = default for all users
    LandingPageAsset NVARCHAR(100) NOT NULL, -- "landing-pages/lc-hollywood"
    CtaAssetId INT NOT NULL,                 -- 10=Soft Follow, 11=Home Value, etc.
    CtaFunction NVARCHAR(50) NOT NULL,       -- "follow", "phone", "email", "full-form"
    TriggerType NVARCHAR(50) DEFAULT 'scroll', -- "immediate", "delay", "scroll", "exit"
    TriggerValue INT DEFAULT 50,            -- seconds for delay, % for scroll
    ShowMobileBanner BIT DEFAULT 1,
    IsActive BIT DEFAULT 1,
    CreateDate DATETIME DEFAULT GETDATE(),
    ModifiedBy NVARCHAR(128)
);

-- CTA Asset definitions (what they look like)
CREATE TABLE FarmGenie.dbo.CtaAsset (
    CtaAssetId INT PRIMARY KEY,
    Name NVARCHAR(100) NOT NULL,
    Description NVARCHAR(255),
    FrictionLevel NVARCHAR(10),              -- "LOW", "MEDIUM", "HIGH"
    TargetEngagementMin INT,
    TargetEngagementMax INT,
    TemplatePath NVARCHAR(200),              -- XSL/JSX template location
    IsActive BIT DEFAULT 1
);
