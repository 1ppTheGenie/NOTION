📜 Stored Procedure: usp_GetNextPlsNumber
CREATE PROCEDURE dbo.usp_GetNextPlsNumber AS BEGIN DECLARE @Year INT = YEAR(GETDATE()); DECLARE @NextNum INT; BEGIN TRANSACTION; -- Create year entry if doesn't exist IF NOT EXISTS (SELECT 1 FROM PlsNumberSequence WHERE Year = @Year) INSERT INTO PlsNumberSequence (Year, NextNumber) VALUES (@Year, 1); -- Get and increment SELECT @NextNum = NextNumber FROM PlsNumberSequence WHERE Year = @Year; UPDATE PlsNumberSequence SET NextNumber = NextNumber + 1 WHERE Year = @Year; COMMIT; -- Return formatted PLS number SELECT 'PLS-' + CAST(@Year AS VARCHAR) + '-' + RIGHT('00000' + CAST(@NextNum AS VARCHAR), 5) AS PlsNumber; END
